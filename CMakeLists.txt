cmake_minimum_required(VERSION 3.16)
project(disk_inspector VERSION 0.3.0 LANGUAGES CXX)
set(DISK_INSPECTOR_MAX_SKIPPED_ENTRIES 10000 CACHE STRING "Maximum ignored/error details retained per scan")
if(NOT DISK_INSPECTOR_MAX_SKIPPED_ENTRIES MATCHES "^[0-9]+$")
    message(FATAL_ERROR "DISK_INSPECTOR_MAX_SKIPPED_ENTRIES must be a non-negative integer")
endif()
add_executable(disk-inspector src/main.cpp src/scanner.cpp src/server.cpp)
target_compile_features(disk-inspector PRIVATE cxx_std_17)
target_include_directories(disk-inspector PRIVATE src)
target_compile_definitions(disk-inspector PRIVATE DISK_INSPECTOR_WEB_DIR="${CMAKE_CURRENT_BINARY_DIR}/web")
target_compile_definitions(disk-inspector PRIVATE DISK_INSPECTOR_MAX_SKIPPED_ENTRIES=${DISK_INSPECTOR_MAX_SKIPPED_ENTRIES})
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/frontend/dist/index.html")
    add_custom_command(TARGET disk-inspector POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/frontend/dist"
            "${CMAKE_CURRENT_BINARY_DIR}/web")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(scanner-tests tests/scanner_tests.cpp src/scanner.cpp)
    target_compile_features(scanner-tests PRIVATE cxx_std_17)
    target_include_directories(scanner-tests PRIVATE src)
    target_compile_definitions(scanner-tests PRIVATE DISK_INSPECTOR_MAX_SKIPPED_ENTRIES=${DISK_INSPECTOR_MAX_SKIPPED_ENTRIES})
    add_test(NAME scanner-tests COMMAND scanner-tests)
endif()
if(WIN32)
    target_link_libraries(disk-inspector PRIVATE ws2_32 shell32 ole32 uuid)
endif()
if(MSVC)
    target_compile_options(disk-inspector PRIVATE /W4 /utf-8)
    if(BUILD_TESTING)
        target_compile_options(scanner-tests PRIVATE /W4 /utf-8)
    endif()
else()
    target_compile_options(disk-inspector PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
        target_compile_options(scanner-tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
